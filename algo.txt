Function UpdateAffectedVertices(G, T, Dist, Parent,
                                Affected_Del, Affected):

    while Affected_Del has true values do

        for each vertex v where Affected_Del[v] == true in parallel:

            Affected_Del[v] = false

            for each child c of v in the SSSP tree T:
                Dist[c] = INFINITY
                Parent[c] = INVALID
                Affected_Del[c] = true
                Affected[c] = true

    while Affected has true values do

        for each vertex v where Affected[v] == true in parallel:

            Affected[v] = false

            // Case 1: Relax outgoing edges (v -> n)
            for each n in Out(v):

                if Dist[v] != INFINITY AND
                   Dist[n] > Dist[v] + W(v,n):

                    Dist[n] = Dist[v] + W(v,n)
                    Parent[n] = v
                    Affected[n] = true

            // Case 2: Find replacement paths using incoming edges (n -> v)
            for each n in In(v):

                if Dist[n] != INFINITY AND
                   Dist[v] > Dist[n] + W(n,v):

                    Dist[v] = Dist[n] + W(n,v)
                    Parent[v] = n
                    Affected[v] = true
